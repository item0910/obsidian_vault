2. Time函数处理:

```java
// SimpleEasyRuleActionServiceImpl.isOverTime
private Boolean isOverTime(SystemThresholdRulesBo rulesDto){
    // 获取当前规则组的时间map
    Map<Long,Long> ruleIdTimeMap = timeCache.get(rulesDto.getRuleGroupId());
    if(ruleIdTimeMap==null){
        ruleIdTimeMap = new HashMap<>();
        timeCache.put(rulesDto.getRuleGroupId(),ruleIdTimeMap);
    }

    Long startTime = ruleIdTimeMap.get(rulesDto.getRuleId());
    if(startTime==null){
        // 首次访问，记录开始时间
        ruleIdTimeMap.put(rulesDto.getRuleId(),System.currentTimeMillis());
    }else{
        // 计算时间差
        Long endTime = System.currentTimeMillis();
        Float timeDiff = (endTime - startTime)/1000F;
        if(timeDiff > Integer.valueOf(rulesDto.getThresholds())){
            return true;
        }
    }
    return false;
}
```

4. longRate(长时变化率)
```java
// 计算公式: |当前值 - 队列尾部值| / (时间间隔 * 长期计数)
String newKey = key.replace(EasyRuleCommonConstants.LONG_RATE_SUFFIX, "");
double abs = 0;
if (longRateCount < 0) {
    Map<String, Float> longRateQueueTailNode = longRateQueue.pollLast();
    if (longRateQueueTailNode != null) {
        abs = Math.abs(currentPiDataMap.get(newKey) - longRateQueueTailNode.get(newKey));
    }
}
BigDecimal bd1 = new BigDecimal(Double.toString(abs));
// 长期变化率时间间隔 = 60 * 5s = 300s (5分钟)
```
