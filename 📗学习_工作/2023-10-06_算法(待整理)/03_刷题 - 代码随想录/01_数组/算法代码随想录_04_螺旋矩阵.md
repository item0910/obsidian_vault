# 59 螺旋矩阵

```java
class Solution {
    public int[][] generateMatrix(int n) {
        int[][] arr = new int[n][n];
        int layer = n >>> 1;
        layer = (n & 1) == 1 ? layer + 1 : layer; 
        int num = 1;
        for(int i = 0; i < layer; i++){
            for(int x = i, y = i; y <= n - i - 2; y++){
                arr[x][y] = num++;
            }
            for(int x = i, y = n - i - 1; x <= n - i - 2; x++){
                arr[x][y] = num++;
            }
            for(int x = n - i - 1, y = n - i - 1; y >= i + 1; y--){
                arr[x][y] = num++;
            }
            for(int x = n - i - 1, y = i ; x >= i + 1; x--){
                arr[x][y] = num++;
            }
        }
        if((n & 1) == 1){
        arr[layer - 1][layer - 1] = num;}
        return arr;
    }
}
```

```java
//一种更优雅的写法
public int[][] generateMatrix(int n) {
	int[][] arr = new int[n][n];
	int c = 1, j = 0;
	while (c <= n * n) {
	
		for (int i = j; i < n - j; i++)
			arr[j][i] = c++;
		for (int i = j + 1; i < n - j; i++)
			arr[i][n - j - 1] = c++;
		for (int i = n - j - 2; i >= j; i--)
			arr[n - j - 1][i] = c++;
		for (int i = n -j - 2; i > j; i--)
			arr[i][j] = c++;

		j++;
	}

	return arr;
}
```
