# 704 二分查找

```java
class Solution {
    public int search(int[] nums, int target) {
        if ((nums == null) || (nums.length == 0)) {
            return -1;
        }

        int length = nums.length;
        int L = 0;
        int R = length - 1;

        while (L <= R) {
            int mid = (L + R) >> 1;

            if (nums[mid] == target) {
                return mid;
            }

            if (nums[mid] > target) {
                R = mid - 1;
            }

            if (nums[mid] < target) {
                L = mid + 1;
            }
        }

        return -1;
    }
}
```

# 34 找到数字在数组中的首尾

```java
class Solution {
    public int[] searchRange(int[] nums, int target) {
        if(nums == null || nums.length == 0){
            return new int[]{-1, -1};
        }
        int left = searchFirstBigger(nums, target - 1);
        int right = searchFirstBigger(nums, target) - 1;
        
        if(left <=right && nums[left] == target){
            return new int[]{left, right};
        }

        return new int[]{-1, -1};
    }
    public int searchFirstBigger(int[] nums, int target){
        int n = nums.length;
        if(nums[n - 1] < target){
            return -1;
        }
        int L = 0; 
        int R = n - 1;
        int position = n;
        while(L <= R){
            int m = (L + R) >>> 1;
            if(nums[m] > target){
                position = m;
                R = m - 1;
            }else{
                L = m + 1;
            }
        }
        return position;
    }
}
```

# 69 求平方根

```java
class Solution {
    public int mySqrt(int x) {
        if(x == 0){
            return 0;
        }
        if(x == 1){
            return 1;
        }
        int L = 1;
        int R = x / 2;
        while(L < R){
            int mid = (L + R + 1) >>> 1;
            if(mid > x / mid){
                R = mid - 1;
            }else{
                L = mid;
            }
        }
        return L;
    }
}

class Solution {
    public boolean isPerfectSquare(int num) {
        if (num == 1) {
            return true;
        }

        int ans = 0;
        int L = 1;
        int R = num;

        while (L < R) {
            int mid = (L + R + 1) >>> 1;

            if (mid > (num / mid)) {
                R = mid - 1;
            } else {
                L = mid;
            }
        }

        ans = L;

        if ((L * L) == num) {
            return true;
        }

        return false;
    }
}
```
