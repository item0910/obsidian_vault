# 面试题 02.07 链表相交

```java
//普通遍历方法
/**
 * Definition for singly-linked list.
 * public class ListNode {
 *     int val;
 *     ListNode next;
 *     ListNode(int x) {
 *         val = x;
 *         next = null;
 *     }
 * }
 */
public class Solution {
    public ListNode getIntersectionNode(ListNode headA, ListNode headB) {
        ListNode dummyA = new ListNode(0);
        ListNode dummyB = new ListNode(0);
        dummyA.next = headA;
        dummyB.next = headB;
        ListNode tempA = dummyA;
        ListNode tempB = dummyB;
        int numA = 0;
        int numB = 0;
        while(tempA.next != null){
            tempA = tempA.next;
            numA++;
        }
        while(tempB.next != null){
            tempB = tempB.next;
            numB++;
        }
        if(tempA != tempB){
            return null;
        }
        int gap = numA > numB ? numA - numB : numB - numA;
        tempA = numA > numB ? dummyA : dummyB;
        tempB = tempA == dummyA ? dummyB : dummyA;
        for(int i = 0; i < gap; i++){
            tempA = tempA.next;
        }
        while(tempA != tempB){
            tempA = tempA.next;
            tempB = tempB.next;
        }
        return tempA;
    }
}
```

//走过相同路程
public class Solution {
    public ListNode getIntersectionNode(ListNode headA, ListNode headB) {
        ListNode A = headA, B = headB;
        while (A != B) {
            A = A != null ? A.next : headB;
            B = B != null ? B.next : headA;
        }
        return A;
    }
}
