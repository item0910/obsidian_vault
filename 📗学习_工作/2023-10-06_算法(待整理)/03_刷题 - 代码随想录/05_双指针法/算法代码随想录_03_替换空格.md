# 剑指 Offer05 替换空格

```java
class Solution {
    public String replaceSpace(String s) {
        char[] arr = s.toCharArray();
        int n = arr.length;

        StringBuilder sb = new StringBuilder();
        for(int i = 0; i < n; i++){
            if(arr[i] == ' '){
                sb.append("%20");
            }else{
                sb.append(arr[i]);
            }
        }
        return new String(sb);
    }
}

//另一个版本, 数组
class Solution {
    public String replaceSpace(String s) {
        char[] arr = s.toCharArray();
        char[] newArr = new char[arr.length * 3];

        int size = 0;
        for(int i = 0; i < arr.length; i++){
            char c = arr[i];
            if(c == ' '){
                newArr[size++] = '%';
                newArr[size++] = '2';
                newArr[size++] = '0';
            }
            else{
                newArr[size++] = c;
            }
        }
        return new String(newArr,0, size);
    }
}
```
