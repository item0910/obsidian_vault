class Solution {
    int num = 0;
    public int findContentChildren(int[] g, int[] s) {
        Arrays.sort(g);
        Arrays.sort(s);
        int count = 0;
        int childrenNum = g.length;
        int cookieNum = s.length;
        for(int i = 0, j = 0; i < childrenNum && j < cookieNum; i++, j++){
            while( j < cookieNum && g[i] > s[j]){
                j++;
            }
            if(j < cookieNum){
                count++;
            }
        }
        return count;      
    }
}
