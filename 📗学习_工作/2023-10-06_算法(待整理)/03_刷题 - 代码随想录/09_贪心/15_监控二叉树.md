# 968.监控二叉树

```java
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    int result = 0;
    public int minCameraCover(TreeNode root) {
        if(post(root) == 0){
            result++;
        }
        return result;
    }

    public int post(TreeNode cur){
        if(cur == null){
            return 2;
        }
        int left = post(cur.left);
        int right = post(cur.right);
        
        if(left == 2 && right == 2){
            //两边都是被覆盖的情况
            return 0;
        }else{
            if(left == 0 || right == 0){
                //如果有一边没有被覆盖
                result++;
                return 1;
            }
            else{
                //没有0, 也不是都是2, 那么必然有一边有1, 执行最下面一行
            }
        }
        return 2;
    }
}
```
