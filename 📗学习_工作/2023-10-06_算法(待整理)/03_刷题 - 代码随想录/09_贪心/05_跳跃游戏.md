# 55.跳跃游戏

```java
class Solution {
    public boolean canJump(int[] nums) {
        if(nums.length == 1){
            return true;
        }
        int cover = 0;
		//注意这里是cover
        for(int i = 0; i <= cover; i++){
            cover = Math.max(i + nums[i], cover);
            if(cover >= nums.length - 1){
                return true;
            }
        }
        return false;
    }
}
```

# 45. 跳跃游戏 II

```java
class Solution {
    public int jump(int[] nums) {
        int step = 0;
        if(nums.length == 1){
            return step;
        }
        int curDistance = 0;
        int nextDistance = 0;
        for(int i = 0; i < nums.length; i++){
            nextDistance = Math.max(nums[i] + i, nextDistance);
            if(i == curDistance){//到达最远处
                if(i < nums.length - 1){//还没到
                    step++;
                    curDistance = nextDistance;
                    if(nextDistance >= nums.length - 1){
                        break;
                    }
                }else{//到了(隐含条件, i == nums.length - 1)
                    break;
                }
            }
        }
        return step;
    }
}
// 版本二
class Solution {
    public int jump(int[] nums) {
        int result = 0;
        // 当前覆盖的最远距离下标
        int end = 0;
        // 下一步覆盖的最远距离下标
        int temp = 0;
        for (int i = 0; i <= end && end < nums.length - 1; ++i) {
            temp = Math.max(temp, i + nums[i]);
            // 可达位置的改变次数就是跳跃次数
            if (i == end) {
                end = temp;
                result++;
            }
        }
        return result;
    }
}
```
