# 112.路径总和

```java
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    public boolean hasPathSum(TreeNode root, int targetSum) {
        if(root == null){
            return false;
        }
        List<Integer> sumPathList = new ArrayList<>();
        List<Integer> list = new ArrayList<>();
        travel(root, list, sumPathList);
        return sumPathList.contains(targetSum);
    }
    public void travel(TreeNode node, List<Integer> list, List<Integer> sumPathList){
        list.add(node.val);
        if(node.left == null && node.right == null){
            int sum = 0;
            for(int i : list){
                sum += i;
            }
            sumPathList.add(sum);
        }
        if(node.left != null){
            travel(node.left, list, sumPathList);
            list.remove(list.size() - 1);
        }
        if(node.right != null){
            travel(node.right, list, sumPathList);
            list.remove(list.size() - 1);
        }
    }
}
```

```java
// lc112 简洁方法
class solution {
    public boolean haspathsum(treenode root, int targetsum) {
        
        if (root == null) return false; // 为空退出
        
        // 叶子节点判断是否符合
        if (root.left == null && root.right == null) return root.val == targetsum;

        // 求两侧分支的路径和
        return haspathsum(root.left, targetsum - root.val) || haspathsum(root.right, targetsum - root.val);
    }
}
```

```java
//用栈迭代
class Solution {
    public boolean hasPathSum(TreeNode root, int targetSum) {
        if(root == null){
            return false;
        }
        Stack<Object> stack = new Stack<>();
        stack.push(root);
        stack.push(root.val);
        while(!stack.isEmpty()){
            int num = (Integer)stack.pop();
            TreeNode node = (TreeNode)stack.pop();
            if(node.left == null && node.right == null && num == targetSum){
                return true;
            }
            if(node.right != null){
                stack.push(node.right);
                stack.push(node.right.val + num);
            }
            if(node.left != null){
                stack.push(node.left);
                stack.push(node.left.val + num);
            }
        }
        return false;
    }
}
```

# 113.路径总和 ii

```java
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    public List<List<Integer>> pathSum(TreeNode root, int targetSum) {
        List<List<Integer>> res = new ArrayList<>();
        if (root == null) return res; // 非空判断
        
        List<Integer> path = new LinkedList<>();
        preorderdfs(root, targetSum, res, path);
        return res;
    }

    public void preorderdfs(TreeNode root, int targetsum, List<List<Integer>> res, List<Integer> path) {
        path.add(root.val);
        // 遇到了叶子节点
        if (root.left == null && root.right == null) {
            // 找到了和为 targetsum 的路径
            if (targetsum - root.val == 0) {
                res.add(new ArrayList<>(path));
            }
            return; // 如果和不为 targetsum，返回
        }

        if (root.left != null) {
            preorderdfs(root.left, targetsum - root.val, res, path);
            path.remove(path.size() - 1); // 回溯
        }
        if (root.right != null) {
            preorderdfs(root.right, targetsum - root.val, res, path);
            path.remove(path.size() - 1); // 回溯
        }
    }
}
```
