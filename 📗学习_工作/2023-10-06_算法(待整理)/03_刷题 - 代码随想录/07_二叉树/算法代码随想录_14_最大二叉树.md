# 654. 最大二叉树

```java
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    public TreeNode constructMaximumBinaryTree(int[] nums) {
        return travel(nums, 0, nums.length);
    }

    public TreeNode travel(int[] nums, int left, int right){
        if(left == right){
            return null;
        }
        int maxIndex = getMaxIndex(nums, left, right);
        TreeNode root = new TreeNode(nums[maxIndex]);
        root.left = travel(nums, left, maxIndex);
        root.right = travel(nums, maxIndex + 1, right);
        return root;
    }

    public int getMaxIndex(int[] nums, int left, int right){
        int res = left;
        for(int i = left; i < right; i++){
            if(nums[i] > nums[res]){
                res = i;
            }
        }
        return res;
    }
}
```
