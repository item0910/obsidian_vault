# 513. 找树左下角的值

```java
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    public int findBottomLeftValue(TreeNode root) {
        Queue<TreeNode> que = new LinkedList<>();
        que.add(root);
        TreeNode res = new TreeNode();
        while(!que.isEmpty()){
            res = que.poll();
            if(res.right != null){
                que.offer(res.right);
            }
            if(res.left != null){
                que.offer(res.left);
            }
        }
        return res.val;
    }
}

//递归
class Solution {
    int maxLevel = -1;
    int res = 0;
    public int findBottomLeftValue(TreeNode root) {
        dfs(root, 0);
        return res;
    }

    public void dfs(TreeNode node, int level){
        if(node == null){
            return;
        }
        //这里注意, 因为在同一层的右边的已经被左边更新过, 所以该逻辑对右节点不再执行
        if(level > maxLevel){
            res = node.val;
            maxLevel = level;
        }
        dfs(node.left, level + 1);
        dfs(node.right, level + 1);
    }
}
```
