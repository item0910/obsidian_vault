# 236. 二叉树的最近公共祖先

```java
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode(int x) { val = x; }
 * }
 */
class Solution {
    private TreeNode ans;
    
    public Solution(){
        this.ans = null;
    }
    //该函数的意思是, 从root出发, 是否能在左子树或者右子树上找到p或者q.
    public boolean ifContains(TreeNode root, TreeNode p, TreeNode q){
        if(root == null){
            return false;
        }
        boolean left = ifContains(root.left, p, q);
        boolean right = ifContains(root.right, p, q);
        if((left && right) || ((root.val == p.val || root.val == q.val) && (left || right))){
            ans = root;
        }

        return left || right || root.val == p.val || root.val == q.val;
    }
    
    public TreeNode lowestCommonAncestor(TreeNode root, TreeNode p, TreeNode q) {
        ifContains(root, p, q);
        return this.ans;
    }
}
```

```java
//记录父节点的方式
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode(int x) { val = x; }
 * }
 */
class Solution {
	
    public TreeNode lowestCommonAncestor(TreeNode root, TreeNode p, TreeNode q) {
        Map<Integer, TreeNode> parentMap = new HashMap<>();
        Set<TreeNode> route = new HashSet<>();
        remarkParent(root, parentMap);
        while(p != null){
            route.add(p);
            p = parentMap.get(p.val);
        }
        while(q != null){
            if(route.contains(q)){
                return q;
            }
            q = parentMap.get(q.val);
        }
        return q;
    }

    public void remarkParent(TreeNode root, Map<Integer, TreeNode> map){
        if(root == null){
            return;
        }
        if(root.left != null){
            map.put(root.left.val, root);
            remarkParent(root.left, map);
        }
        if(root.right != null){
            map.put(root.right.val, root);
            remarkParent(root.right, map);
        }
    } 
}
```

# 235. 二叉搜索树的最近公共祖先

```java
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode(int x) { val = x; }
 * }
 */
//传统方法
class Solution {
    private TreeNode res;

    public TreeNode lowestCommonAncestor(TreeNode root, TreeNode p, TreeNode q) {
        res = null;
        isfind(root, p, q);
        return res;
    }

    public boolean isfind(TreeNode root, TreeNode p, TreeNode q){
        if(root == null){
            return false;
        }
        boolean l = isfind(root.left, p, q);
        boolean r = isfind(root.right, p, q);
        if((l && r) || ((root.val == p.val || root.val == q.val) && (l || r))){
            res = root;
        }

        return l || r || root.val == p.val || root.val == q.val;
    }
}
```

```java
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode(int x) { val = x; }
 * }
 */
//死板递归
class Solution {

    private TreeNode res = null;

    public TreeNode lowestCommonAncestor(TreeNode root, TreeNode p, TreeNode q) {
        dfs(root, p, q);
        return res;
    }

    public void dfs(TreeNode root,TreeNode p,TreeNode q){
        if(root == null || res != null){
            return;
        }
        int bigger = p.val > q.val ? p.val : q.val;
        int smaller = bigger == p.val ? q.val : p.val;

        dfs(root.left, p, q);
        if(root.val >= smaller && root.val <= bigger){
            res = root;
        }
        dfs(root.right, p, q);
    } 
}
```

```java
class Solution {
    public TreeNode lowestCommonAncestor(TreeNode root, TreeNode p, TreeNode q) {
        if(root.val > p.val && root.val > q.val) return lowestCommonAncestor(root.left, p, q);
        if(root.val < p.val && root.val < q.val) return lowestCommonAncestor(root.right, p, q);
        return root;
    }
}

class Solution {
    public TreeNode lowestCommonAncestor(TreeNode root, TreeNode p, TreeNode q) {
        while(true){
            if(root.val > p.val && root.val > q.val){
                root = root.left;
            }
            else if(root.val < p.val && root.val < q.val){
                root = root.right;
            }else{
                break;
            }
        }
        return root;
    }
}
```
