# 面试题 02.07 链表相交

```java
//普通遍历方法
/**
 * Definition for singly-linked list.
 * public class ListNode {
 *     int val;
 *     ListNode next;
 *     ListNode(int x) {
 *         val = x;
 *         next = null;
 *     }
 * }
 */
public class Solution {
    public ListNode getIntersectionNode(ListNode headA, ListNode headB) {
        ListNode dummyA = new ListNode(0);
        ListNode dummyB = new ListNode(0);
        dummyA.next = headA;
        dummyB.next = headB;
        ListNode tempA = dummyA;
        ListNode tempB = dummyB;
        int numA = 0;
        int numB = 0;
        while(tempA.next != null){
            tempA = tempA.next;
            numA++;
        }
        while(tempB.next != null){
            tempB = tempB.next;
            numB++;
        }
        if(tempA != tempB){
            return null;
        }
        int gap = numA > numB ? numA - numB : numB - numA;
        tempA = numA > numB ? dummyA : dummyB;
        tempB = tempA == dummyA ? dummyB : dummyA;
        for(int i = 0; i < gap; i++){
            tempA = tempA.next;
        }
        while(tempA != tempB){
            tempA = tempA.next;
            tempB = tempB.next;
        }
        return tempA;
    }
}
```

```java
//走过相同路程
public class Solution {
    public ListNode getIntersectionNode(ListNode headA, ListNode headB) {
        ListNode A = headA, B = headB;
        while (A != B) {
            A = A != null ? A.next : headB;
            B = B != null ? B.next : headA;
        }
        return A;
    }
}
```

# 142 环形链表的入环点

```java
//O(N)空间的解法
public class Solution {
    public ListNode detectCycle(ListNode head) {
        ListNode pos = head;
        Set<ListNode> visited = new HashSet<ListNode>();
        while (pos != null) {
            if (visited.contains(pos)) {
                return pos;
            } else {
                visited.add(pos);
            }
            pos = pos.next;
        }
        return null;
    }
}
```

```java
//快慢指针的解法
/**
 * Definition for singly-linked list.
 * class ListNode {
 *     int val;
 *     ListNode next;
 *     ListNode(int x) {
 *         val = x;
 *         next = null;
 *     }
 * }
 */
public class Solution {
    public ListNode detectCycle(ListNode head) {
        if(head == null || head.next == null){
            return null;
        }
        ListNode dummy = new ListNode(0);
        dummy.next = head;
        ListNode fast = dummy;
        ListNode slow = dummy;
        while(fast.next != null && fast.next.next != null){
            slow = slow.next;
            fast = fast.next.next;
            if(slow == fast){
                break;
            }
        }
        if(slow != fast){
            return null;
        }
        fast = dummy;
        while(slow != fast){
            slow = slow.next;
            fast = fast.next;
        }
        return fast;
    }
}
```
