# 93. 复原 IP 地址

```java
class Solution {
    List<String> res = new ArrayList<>();
    StringBuilder path = new StringBuilder();

    public List<String> restoreIpAddresses(String s) {
        if(s.length() < 4 || s.length() > 12){
            return res;
        }
        restoreIpAddressesHandler(s, 0, 0);
        return res;
    }

    public void restoreIpAddressesHandler(String s, int startIndex, int number){
        //符合要求
        if(startIndex == s.length() && number == 4){
            res.add(path.toString());
            return;
        }
        //不符合要求
        if(startIndex == s.length() || number == 4){
            return;
        }
        for(int i = startIndex; i < s.length() && i - startIndex < 3; i++){
            if(isValid(s, startIndex, i)){
                if(number < 3){
                    path.append(s.substring(startIndex, i + 1));
                    path.append(".");
                }else{
                    path.append(s.substring(startIndex, i + 1));
                }
            }else{
                continue;
            }
            number++;
            restoreIpAddressesHandler(s, i + 1, number);
            number--;
            if(number < 3){
                //这里要注意要把标点也算进去(path和s下标不同)
                path.delete(startIndex + number, number + i + 2);
            }else{
                path.delete(startIndex + number, number + i + 1);
            }
        }

    }

    public boolean isValid(String s, int start, int end){
        if(start > end){
            return false;
        }
        if(s.charAt(start) == '0' && start != end){
            //start == end说明只有一个数字, 是合法的
            return false;
        }
        int num = 0;
        for(int i = start; i <= end; i++){
            if(s.charAt(i) > '9' || s.charAt(i) < '0'){
                return false;
            }
            num = num * 10 + (s.charAt(i) - '0');
            if(num > 255){
                return false;
            }
        }
        return true;
    }
}
```
