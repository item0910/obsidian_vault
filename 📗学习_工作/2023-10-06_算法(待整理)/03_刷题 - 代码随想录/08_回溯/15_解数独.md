class Solution {
    public void solveSudoku(char[][] board) {
        shuduHandler(board);
    }

    public boolean shuduHandler(char[][] board){
        for(int i = 0; i < 9; i++){
            for(int j = 0; j < 9; j++){
                if(board[i][j] != '.'){
                    continue;
                }
                for(char c = '1'; c <= '9'; c++){
                    if(isValid(i, j, c, board)){
                        board[i][j] = c;
                        if(shuduHandler(board)){
                            return true;
                        }
                        board[i][j] = '.';
                    }
                    
                }
				//1 - 9 都没找到, return false
                return false;
            }
        }
		//全部continue了, return true
        return true;
    }

    public boolean isValid(int row, int col, char c, char[][] board){
        for(int i = 0; i < 9; i++){
            if(board[row][i] == c){
                return false;
            }
        }

        for(int j = 0; j < 9; j++){
            if(board[j][col] == c){
                return false;
            }
        }
        int startRow = (row / 3) * 3;
        int startCol = (col / 3) * 3;
        for(int i = startRow; i < startRow + 3; i++){
            for(int j = startCol; j < startCol + 3; j++){
                if(board[i][j] == c){
                    return false;
                }
            }
        }
        return true;
    }
}
