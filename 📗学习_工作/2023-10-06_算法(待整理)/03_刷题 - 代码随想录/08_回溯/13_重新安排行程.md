# 332.重新安排行程

本题在数据结构和算法上都值得花心思看

```java
class Solution {
    private LinkedList<String> res;
    private Map<String, Map<String, Integer>> ticketMap;
    
    public List<String> findItinerary(List<List<String>> tickets) {
        ticketMap = new HashMap<String, Map<String, Integer>>();
        res = new LinkedList<>();
        for(List<String> ticket: tickets){
			//整理票
            Map<String, Integer> ticketCountMap;
            if(ticketMap.containsKey(ticket.get(0))){
                ticketCountMap = ticketMap.get(ticket.get(0));
                ticketCountMap.put(ticket.get(1), ticketCountMap.getOrDefault(ticket.get(1), 0) + 1);
            }else{
				//升序map
                ticketCountMap = new TreeMap<>();
                ticketCountMap.put(ticket.get(1), 1);
            }
            ticketMap.put(ticket.get(0), ticketCountMap);
        }
        res.add("JFK");
        backTracking(tickets.size());
        return new ArrayList<>(res);
    }

    public boolean backTracking(int ticketNum){
        if(res.size() == ticketNum + 1){
            return true;
        }
        String last = res.getLast();
        if(ticketMap.containsKey(last)){
            for(Map.Entry<String, Integer> target : ticketMap.get(last).entrySet()){
                int count = target.getValue();
                if(count > 0){
                    res.add(target.getKey());
                    target.setValue(count - 1);
                    if(backTracking(ticketNum)) return true;
                    res.removeLast();
                    target.setValue(count);
                }
            }
        }
        return false;
    }
}
```
