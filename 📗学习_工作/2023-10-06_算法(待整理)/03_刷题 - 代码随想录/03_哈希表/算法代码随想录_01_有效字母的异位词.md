# 242 有效的字母异位词

给定两个字符串 s 和 t ，编写一个函数来判断 t 是否是 s 的字母异位词。

示例 1: 输入: s = "anagram", t = "nagaram" 输出: true

示例 2: 输入: s = "rat", t = "car" 输出: false

**说明:** 你可以假设字符串只包含小写字母。

class Solution {
    public boolean isAnagram(String s, String t) {
        if(s.length() != t.length())
            return false;
        int[] alpha = new int[26];
        for(int i = 0; i< s.length(); i++) {
            alpha[s.charAt(i) - 'a'] ++;
            alpha[t.charAt(i) - 'a'] --;
        }
        for(int i=0;i<26;i++)
            if(alpha[i] != 0)
                return false;
        return true;
    }
}

class Solution {
    public List<List<String>> groupAnagrams(String[] strs) {
        HashMap<String, List<String>> map = new HashMap<>();
        for(String s : strs){
            char[] str = s.toCharArray();
			//这里是计划用排序后的数组生成的字符串做key
			//也可以用字频数组做key;
            Arrays.sort(str);
            String key = new String(str);
			//注意这个函数的用法
            List<String> list = map.getOrDefault(key, new ArrayList<String>());
            list.add(s);
            map.put(key,list);
        }
		//注意这里的函数用法
        return new ArrayList<List<String>>(map.values());
    }
}

class Solution {
    public List<Integer> findAnagrams(String s, String p) {
        int n = s.length(), m = p.length();
        List<Integer> res = new ArrayList<>();
        if(n < m) return res;

        int[] pCnt = new int[26];
        int[] sCnt = new int[26];

        for(int i = 0; i < m; i++){
            pCnt[p.charAt(i) - 'a'] ++;
        }
        int L = 0;
        int R = 0;
        while(R < s.length()){
            sCnt[s.charAt(R) -'a']++;
            if(R - L + 1 > p.length()){
                sCnt[s.charAt(L++) - 'a']--;
            }
            if(R - L + 1 == p.length() && isSame(sCnt, pCnt)){
                res.add(L);
            }
            R++;
        }
              
        return res;
    }

    public boolean isSame(int[] a, int[] b){
        for(int i = 0; i < a.length; i++){
            if(a[i] != b[i]){
                return false;
            }
        }
        return true;
    }
}

//双指针法:
class Solution {
    public List<Integer> findAnagrams(String s, String p) {
        int n = s.length(), m = p.length();
        List<Integer> res = new ArrayList<>();
        if(n < m) return res;

        int[] pCnt = new int[26];
        int[] sCnt = new int[26];

        for(int i = 0; i < m; i++){
            pCnt[p.charAt(i) - 'a'] ++;
        }

        int left = 0;
        for(int right = 0; right < n; right++){
            int curRight = s.charAt(right) - 'a';
            sCnt[curRight]++;
			//正常情况词频相等, 这个循环不会进入, 出现未出现
			//的字符时, 通过移动left来对其词频进行删除.
            while(sCnt[curRight] > pCnt[curRight]){
                int curLeft = s.charAt(left) - 'a';
                sCnt[curLeft]--;
                left++;
            }
            if(right - left + 1 == m){
                res.add(left);
            }
        }
        return res;
    }
}
