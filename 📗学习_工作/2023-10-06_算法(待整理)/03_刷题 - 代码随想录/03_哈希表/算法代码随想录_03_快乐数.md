# 202 快乐数

```java
//用set来防止循环.
class Solution {
    public boolean isHappy(int n) {
        if(n == 1){
            return true;
        }
        HashSet<Integer> set = new HashSet<>();
        int sum = 0;
        while(!set.contains(n)){
            set.add(n);
            while(n != 0){
                int rest = n % 10;
                n = n / 10;
                sum += rest * rest;
            }
            if(sum == 1){
                return true;
            }else{
                n = sum;
                sum = 0;
            }
        }
        return false;
    }
}
```

```java
//用快慢指针的思维
class Solution {
    public boolean isHappy(int n) {
        int slow = n;
        int fast = n;
        do{
            slow = getSum(slow);
            fast = getSum(fast);
            fast = getSum(fast);
        }while(slow != fast);
        if(slow == 1){
            return true;
        }
        return false;
    }

    public int getSum(int n){
        int sum = 0;
        while(n != 0){
            int res = n % 10;
            sum += res * res;
            n = n / 10; 
        }
        return sum;
    }
}
```
