# 1049.最后一块石头的重量

```java
//根据递归推出的动归
//递归
class Solution {
    public int lastStoneWeightII(int[] stones) {
        int sum = 0;
        for(int stone : stones){
            sum += stone;
        }
        int target = sum / 2;
        int max = putBag(stones, 0, target);
        return Math.abs(sum - 2 * max);
    }

    public int putBag(int[] stones, int index, int target){
        if(index == stones.length){
            return 0;
        }
        int p1 = putBag(stones, index + 1, target);
        int p2 = Integer.MIN_VALUE;
        if(target >= stones[index]){
            p2 = stones[index] + putBag(stones, index + 1, target - stones[index]);
        }
        return Math.max(p1, p2);
    }
}
//动归
class Solution {
    public int lastStoneWeightII(int[] stones) {
        int sum = 0;
        for(int stone : stones){
            sum += stone;
        }
        int target = sum / 2;
        int n = stones.length;
        int[][] dp = new int[n + 1][target + 1];
        for(int i = 0; i <= target; i++){
            dp[n][i] = 0;
        }
        for(int index = n - 1; index >= 0; index--){
            for(int rest = 0; rest <= target; rest++){
                int p1 = dp[index + 1][rest];
                int p2 = Integer.MIN_VALUE;
                if(rest >= stones[index]){
                    p2 = stones[index] + dp[index + 1][rest - stones[index]];
                }
                dp[index][rest] = Math.max(p1, p2);
            }
        }
        return Math.abs(sum - 2 * dp[0][target]);
    }
}
```
