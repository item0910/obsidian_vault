# 70.爬楼梯

```java
//傻缓存方法
class Solution {
    Map<Integer, Integer> map = new HashMap<>();

    public int climbStairs(int n) {
        return climb(0, n);
    }

    public int climb(int start, int n){
        if(start == n){
            return 1;
        }
        if(start > n){
            return 0;
        }
        int res = 0;
        if(map.containsKey(start)){
            res = map.get(start);
        }else{
            res = climb(start + 1, n) + climb(start + 2, n);
            map.put(start, res);
        }
        return res;
    }
}
```

class Solution {
	//递归出的动态规划
    public int climbStairs(int n) {
        int[] dp = new int[n + 2];
        dp[n] = 1;
        dp[n + 1] = 0;
        for(int i = n - 1; i >= 0; i--){
            dp[i] = dp[i + 1] + dp[i + 2];
        }
        return dp[0];
    }
}
