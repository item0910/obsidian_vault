# 494.目标和

```java
//递归
class Solution {
    public int findTargetSumWays(int[] nums, int target) {
        return getTarget(nums, 0, target);
    }

    public int getTarget(int[] nums, int index, int rest){
        if(index == nums.length){
            if(rest == 0){
                return 1;
            }else{
                return 0;
            }
        }
        return getTarget(nums, index + 1, rest + nums[index]) + getTarget(nums, index + 1, rest - nums[index]);
    }
}
```

```java
//加个傻缓存
class Solution {
    Map<String, Integer> map = new HashMap<>();
    public int findTargetSumWays(int[] nums, int target) {
        return getTarget(nums, 0, target);
    }

    public int getTarget(int[] nums, int index, int rest){
        if(index == nums.length){
            if(rest == 0){
                return 1;
            }else{
                return 0;
            }
        }
        String key = String.valueOf(index) + "," + String.valueOf(rest);
        if(map.containsKey(key)){
            return map.get(key);
        }
        int value = getTarget(nums, index + 1, rest + nums[index]) + getTarget(nums, index + 1, rest - nums[index]);
        map.put(key, value);
        return value;
    }
}
```
