# 62.不同路径

```java
根据递归改动态规划
/*
原始递归
class Solution {
    
    public int uniquePaths(int m, int n) {
        return path(0, 0, m, n);
    }

    public int path(int startX, int startY, int m, int n){
        if(startX == m - 1 && startY == n - 1){
            return 1;
        }
        if(startX == m - 1){
            return path(startX, startY + 1, m, n);
        }
        if(startY == n - 1){
            return path(startX + 1, startY, m, n);
        }
        return path(startX + 1, startY, m, n) + path(startX, startY, m, n);
    }
}
*/

class Solution { 
    public int uniquePaths(int m, int n) {
        int[][] dp = new int[m][n];
        for(int i = m - 1; i >= 0; i--){
            for(int j = n - 1; j >= 0; j--){
                if(i == m - 1 && j == n - 1){
                    dp[i][j] = 1;
                }else if(i == m - 1){
                    dp[i][j] = dp[i][j + 1];
                }else if(j == n - 1){
                    dp[i][j] = dp[i + 1][j];
                }else{
                    dp[i][j] = dp[i + 1][j] + dp[i][j + 1];
                }
            }
        }
        return dp[0][0];
    }
}
```
