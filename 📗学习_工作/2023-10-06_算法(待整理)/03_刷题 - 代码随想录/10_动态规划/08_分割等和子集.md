# 416. 分割等和子集

```java
/**
 * @author Item
 */
//根据递归找动态规划
public class Code0416 {
    public static boolean canPartition(int[] nums) {
        int sum = 0;
        for (int num : nums) {
            sum += num;
        }
        if (sum % 2 != 0) {
            return false;
        }
        int n = nums.length;
        int target = sum / 2;

        boolean[][] dp = new boolean[n + 1][target + 1];
        dp[n][0] = true;
        for (int i = 1; i <= target; i++) {
            dp[n][i] = false;
        }
        for (int index = n - 1; index >= 0; index--) {
            for (int rest = 1; rest <= target; rest++) {
                boolean p1 = dp[index + 1][rest];
                boolean p2 = false;
                if (rest >= nums[index]) {
                    p2 = dp[index + 1][rest - nums[index]];
                }
                dp[index][rest] = p1 || p2;
            }
        }
        return dp[0][target];
        //return canPart(nums, 0, target);
    }

    public static boolean canPart(int[] nums, int index, int rest) {
        if (index == nums.length) {
            if (rest == 0) {
                return true;
            } else {
                return false;
            }
        }
        boolean p1 = canPart(nums, index + 1, rest);
        boolean p2 = false;
        if (rest >= nums[index]) {
            p2 = canPart(nums, index + 1, rest - nums[index]);
        }
        return p1 || p2;
    }

    public static void main(String[] args) {
        int[] test = new int[]{1,5,11,5};
        boolean flag = canPartition(test);
        System.out.println(flag);
    }
}
```
