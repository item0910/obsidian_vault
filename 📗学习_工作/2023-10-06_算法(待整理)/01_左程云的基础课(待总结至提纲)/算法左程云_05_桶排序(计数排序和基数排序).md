```java
public class Code01_Trie {
	public static class Node{
		int pass;
		int end;
		public Node[] nexts;
		
		public Node(){
			pass = 0;
			end = 0;
			nexts = new Node[26];
		}
	}
	
	public static class Trie{
		private Node root;
		
		public Trie(){
			root = new Node();
		}
		
		public void insert(String s){
			if(s == null){
				throw new RuntimeException("字符串为空");
			}
			if(search(s)){
				return;
			}
			char[] str = s.toCharArray();
			Node node = root;
			node.pass++;
			int path = 0;
			for(int i = 0; i < str.length; i++){
				path = str[i] - 'a';
				if(node.nexts[path] == null){
					node.nexts[path] = new Node();
				}
				node = node.nexts[path];
				node.pass++;
			}
			node.end++;
		}
		
		public boolean search(String s){
			if(s == null){
				throw new RuntimeException("字符串为空");
			}
			char[] str = s.toCharArray();
			Node node = root;
			int path = 0;
			for(int i = 0; i < str.length; i++){
				path = str[i] - 'a';
				if(node.nexts[path] == null){
					return false;
				}
				node = node.nexts[path];
			}
			
			return node.end == 1;
		}
		

		
		public int prefixNum(String pre){
			if(pre == null){
				throw new RuntimeException("字符串为空");
			}
			char[] str = pre.toCharArray();
			Node node = root;
			int path = 0;
			for(int i = 0; i < str.length; i++){
				path = str[i] - 'a';
				if(node.nexts[path] == null){
					return 0;
				}
				node = node.nexts[path];
			}
			return node.pass;
		}
		
		public void remove(String s){
			if(search(s)){
				char[] str = s.toCharArray();
				Node node = root;
				node.pass--;
				int path = 0;
				for(int i = 0; i < str.length; i++){
					path = str[i] - 'a';
					if(--node.nexts[path].pass == 0){
						node.nexts[path] = null;
						return;
					}
					node = node.nexts[path];
				}
				node.end--;
			}
		}
	}

			Node2 node = root;
			node.pass++;
			for(int i = 0; i < str.length; i++){

		}

			}

			char[] str = pre.toCharArray();
			for(int i = 0; i < str.length; i++){
				int ch = (int)(str[i]);
				if(!node.nexts.containsKey(ch)){
					return 0;

				node = node.nexts.get(ch);

				return;

			node.pass--;

				int ch = (int)(str[i]);
				if(--node.nexts.get(ch).pass == 0){
					node.nexts.remove(ch);
					return;
				}
				node = node.nexts.get(ch);
			}
			node.end--;
		}
```

```java
package class005;

public class Code02_CountSort {
	public void CountSort(int[] arr){
		int max = Integer.MIN_VALUE;
		for(int i = 0; i < arr.length; i++){
			max = Math.max(max, arr[i]);
		}
		
		int[] help = new int[max + 1];
		for(int i = 0; i < arr.length; i++){
			help[arr[i]]++;
		}
		for(int i = 0, j = 0; i < help.length; i++){
			while(help[i]-- > 0){
				arr[j++] = i;
			}
		}
	}	
}
```

# 桶排序之基数排序

1. 准备一个一样长度的 help 数组.
2. 将每个数字从个位到首位出现的频次记录在 count[10] 里.
3. count 自处理 (从 1 开始 count[i] + count[i - 1]), 把每项变换为小于等于该位置数的个数.
4. 将原数组从又往左遍历, 从最低位开始观察, 因为本来要放在最右边, 所以会在小于等于的个数的 count[i] - 1 位置.
5. 依次循环过程 4, 直到最高位结束.

```java
public class Code03_RadixSort {
	public static void radixSort(int[] arr){
		if(arr == null || arr.length < 2){
			return;
		}
		radixSort(arr , 0, arr.length - 1, maxbits(arr));
	}
	
	public static int maxbits(int[] arr){
		int max = Integer.MIN_VALUE;
		for(int i = 0; i <arr.length; i++){
			max = Math.max(max, arr[i]);
		}
		int res = 0;
		while(max != 0){
			res++;
			max /= 10;
		}
		return res;
	}
	
	public static void radixSort(int[] arr, int L, int R, int digit){
		//radix表示进制;digix表示位数
		final int radix = 10;
		//准备倒筒空间
		int[] help = new int[R - L + 1];
		//准备记录基数频次的空间
		int foot = 0;
		for(int d = 1; d <= digit; d++){
			int[] count = new int[radix];
			for(int i = L; i <= R; i++){
				foot = getFoot(arr[i], d);
				count[foot]++;
			}
			
			//变换count数组
			for(int i = 1; i < count.length; i++){
				count[i] = count[i - 1] + count[i];
			}
			
			//从后往前倒
			for(int i = R; i>= L; i--){
				foot = getFoot(arr[i],d);
				help[count[foot]-1] = arr[i];
				count[foot]--;
			}
			
			for(int i = L, j = 0; i <= R; i++,j++){
				arr[i] = help[j];
			}
		}
		
	}
	
	public static int getFoot(int num, int d){
		return (num /(int) (Math.pow(10, d - 1))) % 10;
	}
	
	public static int[] generateRandomArray(int maxSize, int maxValue) {
		int[] arr = new int[(int) ((maxSize + 1) * Math.random())];
		for (int i = 0; i < arr.length; i++) {
			arr[i] = (int) ((maxValue + 1) * Math.random());
		}
		return arr;
	}
```
