```java
package class008;

public class Code01_MinAppleBags {
    public static int minAppleBags(int appleNum){
		//小于6或者奇数, 直接返回 -1
        if(appleNum < 6 || (appleNum & 1) == 0){
            return -1;
        }
		
		//先求出bag8的个数
        int bag8 = appleNum >> 3;
        int rest = appleNum - (bag8 << 3);
        while(bag8 >= 0){
			//如果剩余能整除
            if(rest % 6 == 0){

            }else{
```

```java
    //先打出来看一下
    public static String eatGrass(int n){
        if(n < 0){
            throw new RuntimeException("有问题");
        }
        if(n < 5 ){
			//简单判断一下1-5的情况
            return (n == 2 || n == 5) ? "后手" : "先手";
        }
        int base = 1;
        while(base <= n){
			//递归,因为在这个角度是先手吃, 那么当"后手胜利时", 返回胜利者为"先手"
```

## 是否可以成为连续正整数的和

```java
package class008;

/**
 * @author Item
 */
public class Code03_ContinusNumSum {
	public static boolean isContinuousNumSum(int num) {
		if (num <= 2) {
			return false;
		}
		//用等差数列遍历
		for (int i = 1; i < num - 1; i++) {
			for (int j = i + 1; j < num; j++) {
				int count = j - i + 1;
				if (num == (i + j) * count / 2) {
					return true;
				}
			}
		}
		return false;
	}

	public static boolean isContinuousNumSum2(int num) {
		//观察结果后,发现只有2的n次方不可以
		//注意2的n次方的表达式
		return (num >= 3) && ((num & (num - 1)) != 0);
	}
	

	public static void main(String[] args) {
		for (int i = 0; i < 1000; i++) {
			System.out.println(i + " : " + isContinuousNumSum2(i));
		}
	}
}
```
