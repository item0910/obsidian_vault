# 类型为 V 的样本, 查是否属于同一个集合:isSameSet, 合并到同一个集合:union

- 先用 Node< V >包装这个类型的对象.
- 再来一个 Union 类, 内里有三个 HashMap
	- nodes< V, node< V >> 存所有 node
	- parents< node< V >, node< V >>存所有 node 的父节点
	- size < node < V >, Integer >存所有父节点的 size

```java
package class010;

import java.util.HashMap;
import java.util.List;
import java.util.Stack;

/**
 * @author Item
 */
public class Code01_UnionFind {
    public static class Node<V> {
        V value;

        public Node(V value) {
            this.value = value;
        }
    }

    public static class Union<V> {
        HashMap<V, Node<V>> nodeMap;
        HashMap<Node<V>, Node<V>> parentMap;
        HashMap<Node<V>, Integer> sizeMap;

        public Union(List<V> list) {
            nodeMap = new HashMap<>();
            parentMap = new HashMap<>();
            sizeMap = new HashMap<>();
            for (V v : list) {
                Node<V> node = new Node<>(v);
                nodeMap.put(v, node);
                parentMap.put(node, node);
                sizeMap.put(node, 1);
            }
        }

        public Node<V> findFather(Node<V> node) {
            Stack<Node<V>> stack = new Stack<>();
            //把所有链上的节点都直接关联到父节点上, 不再走链.
            while (node != parentMap.get(node)) {
                stack.push(node);
                node = parentMap.get(node);
            }
            while (!stack.isEmpty()) {
                parentMap.put(stack.pop(), node);
            }
            return node;
        }

        public boolean isSameSet(V v1, V v2) {
            return findFather(nodeMap.get(v1)) == findFather(nodeMap.get(v2));
        }

        public void union(V a, V b) {
			//注意使用contains方法
            if (!nodeMap.containsKey(a) || !nodeMap.containsKey(b)){
                return;
            }
            Node<V> aParent = findFather(nodeMap.get(a));
            Node<V> bParent = findFather(nodeMap.get(b));
            if(aParent != bParent){
				//短链连长链
                int sizeA = sizeMap.get(aParent);
                int sizeB = sizeMap.get(bParent);

                if(sizeA > sizeB){
                    parentMap.put(nodeMap.get(b),aParent);
                    sizeMap.put(aParent, sizeA + sizeB);
                    sizeMap.remove(bParent);
                }else{
                    parentMap.put(nodeMap.get(a),bParent);
                    sizeMap.put(bParent, sizeA + sizeB);
                    sizeMap.remove(aParent);
                }
            }
        }
    }
}
```

# UnionStudent

- 新建一个方法: int mergeStudents(List< Students> students)
- 方法的内部变量:
	- unionSet
	- 字段与变量的 Map(String, Student)
		- 遍历 list, 如果字段没在 map 登记, 就登记, 如果登记过了, 就 union
	- 最后返回并查集中 sizeMap 的 size;

```java
/**
 * @author Item
 */
public class Code01_UnionStudent {
    public static class Student {
        String a;
        String b;
        String c;

        public Student(String a, String b, String c) {
            this.a = a;
            this.b = b;
            this.c = c;
        }
    }

    public int mergeStudents(List<Student> students) {

        UnionSet<Student> union = new UnionSet<Student>(students);

        HashMap<String, Student> mapA = new HashMap<>();
        HashMap<String, Student> mapB = new HashMap<>();
        HashMap<String, Student> mapC = new HashMap<>();

        for (Student s : students) {
            if (mapA.containsKey(s.a)) {
                union.union(s, mapA.get(s.a));
            } else {
                mapA.put(s.a, s);
            }
            if (mapB.containsKey(s.b)) {
                union.union(s, mapB.get(s.b));
            } else {
                mapB.put(s.b, s);
            }
            if (mapC.containsKey(s.c)) {
                union.union(s, mapC.get(s.c));
            } else {
                mapC.put(s.c, s);
            }
        }
        return union.getSetSize();
    }

    public static class Node<V> {
        V value;

        public Node(V value) {
            this.value = value;
        }
    }

    public static class UnionSet<V> {
        public HashMap<V, Node<V>> nodes;
        public HashMap<Node<V>, Node<V>> parents;
        public HashMap<Node<V>, Integer> sizeMap;

        public UnionSet(List<V> list) {
            nodes = new HashMap<>();
            parents = new HashMap<>();
            sizeMap = new HashMap<>();


            for (V v : list) {
                Node<V> node = new Node<>(v);
                nodes.put(v, node);
                parents.put(node, node);
                sizeMap.put(node, 1);
            }
        }

        public Node<V> findFather(Node<V> node) {
            Stack<Node<V>> stack = new Stack<>();
            while (node != parents.get(node)) {
                stack.push(node);
                node = parents.get(node);
            }
            while (!stack.isEmpty()) {
                parents.put(stack.pop(), node);
            }
            return node;
        }

        public boolean isSameSet(V a, V b) {
            return findFather(nodes.get(a)) == findFather(nodes.get(b));
        }

        public void union(V a, V b) {
            if (!nodes.containsKey(a) || !nodes.containsKey(b)) {
                return;
            }
            Node<V> parentA = findFather(nodes.get(a));
            Node<V> parentB = findFather(nodes.get(b));
            if (parentA != parentB) {
                int sizeA = sizeMap.get(parentA);
                int sizeB = sizeMap.get(parentB);
                Node<V> small = sizeA < sizeB ? parentA : parentB;
                Node<V> big = small == parentA ? parentB : parentA;
                parents.put(small, big);
                sizeMap.put(big, sizeA + sizeB);
                sizeMap.remove(small);
            }
        }
        public int getSetSize(){
            return sizeMap.size();
        }
    }

}

```
