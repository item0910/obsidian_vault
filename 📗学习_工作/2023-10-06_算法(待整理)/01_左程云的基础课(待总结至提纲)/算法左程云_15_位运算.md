```java
package class013;

/**

public class Code02_IntBitMap {
    public static void main(String[] args) {
        int[] arr = new int[10];//32bit * 10 = 320bit
        /**
         *  arr[0] 0 ~ 31
         *  arr[1] 32 ~ 63
         *  arr[2] 64 ~ 95
         */
        int i = 178; //需要判断的状态

        int numIndex = 178 / 32;
        int bitIndex = 178 % 32;

        //拿到这个数字的状态
        int s = (arr[numIndex] >> (bitIndex)) & 1;

        //置1 这个位置的状态
        arr[numIndex] = arr[numIndex] | (1 << bitIndex);

        //置0 这个位置的状态
        arr[numIndex] = arr[numIndex] & (~ (1 << bitIndex));
    }
}

```

# 给定两个有符号 32 位整数 a 和 b, 返回 a 和 b 中比较大的

- 利用互斥的加法

```java
package class013;
/**
 *
 * @author Item
 */
public class Code09_GetMaxW {
    public static int flip(int n){
        return n ^ 1;
    }
    //取出n的符号位, flip: n如果非负, 返回1
    public static int sign(int n){
        return flip((n >> 31) & 1);
    }

    public static int getMax1(int a, int b){
        //这里有越界的问题, 所以有getMax2
        int c = a - b;
        int scA = sign(c);//如果a-b非负就为1, 负就为0
        int scB = flip(scA);
        //互斥相加
        return a * scA + b * scB;
    }

    public static int getMax2(int a, int b){
        int c = a - b;
        int sa = sign(a);
        int sb = sign(b);
        int sc = sign(c);
        int difSab = sa ^ sb;//如果a, b符号不同, 为1; 相同为0
        int sameSab = flip(difSab);
        //什么时候会返回a?
        //情况1: a b 同符号, a - b > 0(sc == 1);
        //情况2: a b 不同符号, a > 0;
        int returnA = difSab * sa + sameSab * sc;
        int returnB = flip(returnA);
        return returnA * a + returnB * b;
    }
}
```

# 判断一个 32 位正数是不是 2 的幂, 4 的幂

- 是否 2 的幂?
	- 判断是否只有一个 1?
	- 1. 拿到最右侧的 1 :
		- int rightOne = num & (~num + 1);
		- if(rightOne == num) 成立
	- 2. x & (x - 1) == 0

- 是否 4 的幂
	- 判断是否只有一个 1 之后, 与 0101010101010101...看是不是为 0, 是的话就是 4 的某次幂.
	- (n & (n - 1) == 0) && (n & 0x55555555) == 0

```java
package class013;

/**
 * @author Item
 */
public class Code10_PlusMinusMultiDivide {
    //如果溢出, 就是溢出.(系统都是错的)
    public static int plus(int a, int b){
        //两数相加结果应该等于异或的结果 + 与的结果 << 1(不进位结果 + 进位)
        int xor = a ^ b;
        int and = (a & b) << 1;

        while(and != 0){
            a = xor;
            b = and;
            xor = a ^ b;
            and = (a & b) << 1;
        }
        return xor;
    }
    //求b的相反数
    public static int negNum(int b){
        return plus(~b, 1);
    }

    public static int minus(int a, int b){
        //a + b的相反数
        return plus(a, negNum(b));
    }

    public static int multi(int a, int b){
        int res = 0;
        while(b != 0){
            if((b & 1) != 0){
                //b末位不是0
                res = plus(res, a);
                //a左移一位, b右移一位.
                a <<= 1;
                b >>>= 1;
            }
        }
        return res;
    }

    public static boolean isNeg(int a){
        return a < 0;
    }

    public static int divide(int a, int b){
        //先用正数做处理, 结尾在来处理符号问题.
        int x = isNeg(a) ? negNum(a) : a;
        int y = isNeg(b) ? negNum(b) : b;
        int res = 0;
        for(int i = 31; i > -1; i = minus(i, 1)){
            if((x >> i) > y){
                res |= (1 << y);
                x = minus(x, y << i);
            }
        }
        return isNeg(a) ^ isNeg(b) ? negNum(res) : res;
    }



    public static void main(String[] args) {
        int time = 1000000;
        int max = 10000;
        int a = (int) (Math.random() * max);
        int b = (int) (Math.random() * max);
```
