## 漫画的处理

- 漫画原版通常是从右往左的阅读顺序, 而且还有跨页的图片, 所以网上下的各种需要处理

### 思路 1, 转换为 cbz 格式:

> 针对网上下的图片文件夹格式的漫画, 有需要进行分页处理的图片

1. 先用 `epub 漫画导出图片.exe` 将 epub 格式 (如果不是需要转) 的文件导出一个图片文件夹 (如果已经是图片文件夹的则不用)
2. 用 `jpeg向上移动一个目录(漫画处理)` 将文件移动到目录下, 然后打开那个 images 的文件夹, 把需要分页的图片放进去
3. 进入图片文件夹看一下顺序, 如果有跨页的, 用 `分页.py` 分一下页 (纵向), 注意 jpeg 和 jpg 格式
4. 用 honeyview 把文件夹先浏览一下, 调整一下顺序
5. 将文件夹压缩为 zip 格式, 放入 calibre, 然后用 honeyview 打开, 选择图片从右到左的阅读模式. yacreader 也不错, 且按键逻辑是从右到左的, 但是好像分页暂时处理不了.

### 思路 2, 转换为 pdf

转换为 pdf 格式之后, 因为也可以用 neeview 打开, 所以可以对 pdf 进行去白边, 加白页面的功能

## 更改阅读方向 (针对 azw3 文件)

```python
import os
import subprocess
import zipfile
from concurrent.futures import ThreadPoolExecutor, as_completed
import shutil

def convert_azw3_to_epub(azw3_path, epub_path):
    """
    使用 Calibre 的 ebook-convert 工具将 .azw3 文件转换为 .epub 文件。

    参数:
    - azw3_path (str): 输入的 .azw3 文件路径。
    - epub_path (str): 输出的 .epub 文件路径。
    """
    result = subprocess.run(['ebook-convert', azw3_path, epub_path], capture_output=True, text=True)
    if result.returncode != 0:
        raise RuntimeError(f"转换 {azw3_path} 到 EPUB 失败: {result.stderr}")

def rename_epub_to_zip_and_extract(epub_path, output_folder):
    """
    将 .epub 文件重命名为 .zip 并解压到指定的文件夹中。

    参数:
    - epub_path (str): 要重命名和解压的 .epub 文件路径。
    - output_folder (str): 解压后的目标文件夹路径。
    """
    zip_path = epub_path.replace('.epub', '.zip')
    os.rename(epub_path, zip_path)  # 重命名为 .zip

    # 解压 .zip 文件
    with zipfile.ZipFile(zip_path, 'r') as zip_ref:
        zip_ref.extractall(output_folder)
    
    os.remove(zip_path)  # 删除临时的 .zip 文件

def replace_content_in_opf(folder_path, replacements):
    """
    在指定文件夹中找到 content.opf 文件并进行内容替换。

    参数:
    - folder_path (str): 包含 content.opf 文件的文件夹路径。
    - replacements (dict): 要进行的替换操作，键为旧内容，值为新内容。
    """
    opf_path = os.path.join(folder_path, 'content.opf')
    if os.path.exists(opf_path):
        with open(opf_path, 'r', encoding='utf-8') as file:
            content = file.read()
        
        for old_text, new_text in replacements.items():
            content = content.replace(old_text, new_text)
        
        with open(opf_path, 'w', encoding='utf-8') as file:
            file.write(content)
    else:
        raise FileNotFoundError(f"未找到 {opf_path}")

def compress_folder_to_epub(folder_path, epub_path):
    """
    将指定文件夹重新压缩为 .epub 文件。

    参数:
    - folder_path (str): 要压缩的文件夹路径。
    - epub_path (str): 输出的 .epub 文件路径。
    """
    with zipfile.ZipFile(epub_path, 'w') as epub_zip:
        for root, _, files in os.walk(folder_path):
            for file in files:
                file_path = os.path.join(root, file)
                # 在 zip 中保持相对路径
                relative_path = os.path.relpath(file_path, folder_path)
                epub_zip.write(file_path, relative_path)
    
    # 确保 mimetype 文件位于 ZIP 的第一位且未压缩
    # 这对于 EPUB 格式的有效性是必要的
    # 重新创建 ZIP 文件以调整顺序
    temp_zip = epub_path + '.temp'
    with zipfile.ZipFile(temp_zip, 'w') as temp_zip_ref:
        # 首先写入 mimetype 文件
        mimetype_path = os.path.join(folder_path, 'mimetype')
        if os.path.exists(mimetype_path):
            with open(mimetype_path, 'rb') as f:
                temp_zip_ref.writestr('mimetype', f.read(), compress_type=zipfile.ZIP_STORED)
        
        # 然后写入其他文件
        for root, _, files in os.walk(folder_path):
            for file in files:
                if file == 'mimetype':
                    continue  # 已处理
                file_path = os.path.join(root, file)
                relative_path = os.path.relpath(file_path, folder_path)
                with open(file_path, 'rb') as f:
                    temp_zip_ref.writestr(relative_path, f.read(), compress_type=zipfile.ZIP_DEFLATED)
    
    # 替换原始 EPUB 文件
    os.remove(epub_path)
    os.rename(temp_zip, epub_path)

def convert_epub_to_azw3(epub_path, azw3_path):
    """
    使用 Calibre 的 ebook-convert 工具将 .epub 文件转换回 .azw3 文件。

    参数:
    - epub_path (str): 输入的 .epub 文件路径。
    - azw3_path (str): 输出的 .azw3 文件路径。
    """
    result = subprocess.run(['ebook-convert', epub_path, azw3_path], capture_output=True, text=True)
    if result.returncode != 0:
        raise RuntimeError(f"转换 {epub_path} 到 AZW3 失败: {result.stderr}")
    os.remove(epub_path)  # 删除临时的 .epub 文件

def process_azw3_file(azw3_path, replacements):
    """
    完整的处理流程：将 .azw3 转换为 .epub，解压，替换内容，重新压缩为 .epub，再转换回 .azw3。

    参数:
    - azw3_path (str): 要处理的 .azw3 文件路径。
    - replacements (dict): 要进行的替换操作，键为旧内容，值为新内容。
    """
    try:
        # 定义路径
        epub_path = azw3_path.replace('.azw3', '.epub')
        output_folder = azw3_path.replace('.azw3', '')
        final_azw3_path = azw3_path.replace('.azw3', '_processed.azw3')

        # 步骤 1: .azw3 转 .epub
        print(f"转换 {azw3_path} 到 EPUB...")
        convert_azw3_to_epub(azw3_path, epub_path)

        # 步骤 2: .epub 改为 .zip 并解压
        print(f"解压 {epub_path} 到 {output_folder}...")
        rename_epub_to_zip_and_extract(epub_path, output_folder)

        # 步骤 3: 替换 content.opf 内容
        print(f"替换 {output_folder}/content.opf 中的内容...")
        replace_content_in_opf(output_folder, replacements)

        # 步骤 4: 压缩文件夹为新的 .epub
        print(f"重新压缩 {output_folder} 为 EPUB...")
        compress_folder_to_epub(output_folder, epub_path)

        # 步骤 5: 将 .epub 转换回 .azw3
        print(f"转换 {epub_path} 到 AZW3...")
        convert_epub_to_azw3(epub_path, final_azw3_path)

        # 步骤 6: 清理临时文件夹
        print(f"清理临时文件夹 {output_folder}...")
        shutil.rmtree(output_folder)

        print(f"处理完成: {final_azw3_path}")

    except Exception as e:
        print(f"处理文件 {azw3_path} 时发生错误: {e}")
        raise  # 重新抛出异常以便在多线程中中断

def process_all_azw3_files(replacements, max_workers=6):
    """
    查找当前目录下的所有 .azw3 文件，使用多线程执行每个文件的完整处理流程。

    参数:
    - replacements (dict): 要进行的替换操作，键为旧内容，值为新内容。
    - max_workers (int): 最大线程数。
    """
    current_dir = os.getcwd()
    azw3_files = [os.path.join(current_dir, f) for f in os.listdir(current_dir) if f.endswith('.azw3')]

    if not azw3_files:
        print("当前目录下没有找到 .azw3 文件。")
        return

    print(f"找到 {len(azw3_files)} 个 .azw3 文件，开始处理...")

    try:
        with ThreadPoolExecutor(max_workers=max_workers) as executor:
            # 提交所有任务
            futures = {executor.submit(process_azw3_file, azw3_file, replacements): azw3_file for azw3_file in azw3_files}

            for future in as_completed(futures):
                azw3_file = futures[future]
                try:
                    future.result()  # 获取结果，若有异常则抛出
                except Exception as e:
                    print(f"处理中断，原因: {e}")
                    executor.shutdown(wait=False)
                    raise  # 中断整个处理流程
    except Exception as e:
        print(f"处理过程中发生异常: {e}")

# 定义需要替换的内容
replacements = {
    'vertical-rl': 'horizontal-lr',
    'rtl': 'ltr'
}

# 运行处理流程
if __name__ == "__main__":
    process_all_azw3_files(replacements)

```
