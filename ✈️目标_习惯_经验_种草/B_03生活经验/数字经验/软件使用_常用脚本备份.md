## 迭代列出目录.py


```python
import os

def list_files_and_dirs():
    # 获取脚本所在目录
    script_dir = os.path.dirname(os.path.abspath(__file__))
    output_file = os.path.join(script_dir, "目录.txt")

    with open(output_file, "w", encoding="utf-8") as f:
        for root, dirs, files in os.walk(script_dir):
            # 输出当前目录
            relative_path = os.path.relpath(root, script_dir)
            f.write(f"目录: {relative_path}\n")
            
            # 输出目录中的子目录
            for d in dirs:
                f.write(f"  子目录: {d}\n")
            
            # 输出目录中的文件
            for file in files:
                f.write(f"  文件: {file}\n")
            
            f.write("\n")
    
    print(f"目录结构已保存到: {output_file}")

if __name__ == "__main__":
    list_files_and_dirs()

```
