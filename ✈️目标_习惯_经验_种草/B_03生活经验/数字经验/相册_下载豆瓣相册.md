## 下载豆瓣相册

1. 来到相册, 比如: [吉冈里帆的图片](https://www.douban.com/personage/27574450/photos/)
2. 打开通过 fatkun 扩展, 获取链接地址: ![](附件/image/数字经验_下载豆瓣相册-1729779708298.jpeg), ![](附件/image/数字经验_下载豆瓣相册-1729779724213.jpeg), 然后把这些地址 down 下来
3. 文本处理, 把解析地址中的 webp 格式的删掉, 只留下 jpg 格式的, 然后把 'photo/photo' 字符, 替换为 'photo/raw 字符 ', 因为这样才能获得原始大图的格式
4. 用事先写好的 py 脚本下载, 注意, 链接的文本要改为 'image_links.txt', 文件名为 ' 批量下载豆瓣相册 '

```python
import concurrent.futures
import requests
import os

# 创建一个文件夹来保存下载的图片
if not os.path.exists('images'):
    os.makedirs('images')

# 从文件读取图片链接
def read_image_links(filename):
    with open(filename, 'r', encoding='utf-8') as file:
        links = [line.split('"')[1] for line in file if 'src=' in line]
    return links

# 模仿浏览器的请求头
headers = {
    "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/85.0.4183.121 Safari/537.36",
    "Referer": "https://www.douban.com/",
    "Accept-Language": "en-US,en;q=0.9",
    "Connection": "keep-alive"
}

# 下载图片的函数
def download_image(url, number):
    try:
        response = requests.get(url, headers=headers)
        if response.status_code == 200:
            img_name = f"{number}.jpg"  # 按顺序命名图片
            with open(f'images/{img_name}', 'wb') as img_file:
                img_file.write(response.content)
            print(f"Downloaded {img_name}")
        else:
            print(f"Failed to download {url}: HTTP {response.status_code}")
    except Exception as e:
        print(f"Error downloading {url}: {e}")

# 批量下载图片
def download_images_in_parallel(links, num_threads=10):
    with concurrent.futures.ThreadPoolExecutor(max_workers=num_threads) as executor:
        executor.map(download_image, links, range(1, len(links) + 1))

if __name__ == "__main__":
    # 读取图片链接
    image_links = read_image_links('image_links.txt')
    
    # 开始并行下载并按序号命名
    download_images_in_parallel(image_links)

```
