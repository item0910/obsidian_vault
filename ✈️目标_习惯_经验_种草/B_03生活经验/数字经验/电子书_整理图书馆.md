### 3. 重命名图书

```python
import os
import re

# 处理文件名：依次执行多个重命名步骤
def clean_book_title(file_name):
    # 提取文件扩展名
    file_name_without_ext, file_ext = os.path.splitext(file_name)

    # 1. 删除套装描述，如 "套装共*册", "共*册", "全*册", "第*卷", "上下册"
    file_name_without_ext = re.sub(r'套装共.*册|全集|共\d册|全\d册|第.*卷|上下册', '', file_name_without_ext)

    # 2. 删除 '(*丛书)'、'(*精选)'、'(*合辑)' 等标识
    file_name_without_ext = re.sub(r'\(.*丛书\)|\(.*精选\)|\(.*合辑\)', '', file_name_without_ext)

    # 3. 删除 '(*版)' 和 '(*卷)' 格式的部分
    file_name_without_ext = re.sub(r'\(.*?版\)|\(.*?卷\)', '', file_name_without_ext)

    # 4. 删除 '-[*]作者名' 或 ' [*]作者名' 部分
    file_name_without_ext = re.sub(r'\s*-\s*\[.*?\].*|\s*\[.*?\].*', '', file_name_without_ext)

    # 5. 删除 "作者**" 开头或中间的部分
    file_name_without_ext = re.sub(r'作者.*', '', file_name_without_ext)

    # 去除多余的空格
    file_name_without_ext = file_name_without_ext.strip()

    # 返回清理后的文件名，并加上原来的扩展名
    return file_name_without_ext + file_ext

# 确保文件名唯一，避免重命名冲突
def get_unique_filename(directory, file_name):
    base_name, file_ext = os.path.splitext(file_name)
    counter = 1
    new_file_name = file_name
    while os.path.exists(os.path.join(directory, new_file_name)):
        new_file_name = f"{base_name} ({counter}){file_ext}"
        counter += 1
    return new_file_name

# 获取当前文件夹中的文件列表并修改文件名
def clean_file_names_in_folder(folder_path):
    files = os.listdir(folder_path)

    # 打开日志文件记录重命名操作
    with open('renaming_log.txt', 'w', encoding='utf-8') as log_file:
        for file_name in files:
            # 处理文件名，执行重命名步骤
            cleaned_name = clean_book_title(file_name)
            
            # 如果清理后的文件名和原文件名不一样，则进行重命名
            if cleaned_name != file_name:
                # 确保新文件名在当前文件夹中唯一
                cleaned_name = get_unique_filename(folder_path, cleaned_name)

                # 构建原文件路径和新文件路径
                original_file = os.path.join(folder_path, file_name)
                cleaned_file = os.path.join(folder_path, cleaned_name)

                # 输出到日志文件
                log_file.write(f"重命名文件: {file_name} -> {cleaned_name}\n")
                print(f"重命名文件: {file_name} -> {cleaned_name}")

                # 重命名文件
                os.rename(original_file, cleaned_file)

# 设置当前文件夹路径
folder_path = '.'  # 当前文件夹

# 执行文件名清理和重命名
clean_file_names_in_folder(folder_path)

```

大概的脚本如上, **注意点**
1. 分步处理
2. 备份, 或者保存日志

### 4. 图书自动分类

- 利用谷歌的 api

3 谷歌匹配.py

```python
import os
import requests
import shutil

# 获取书名（文件名）的函数，保留扩展名，忽略 .py 文件
def get_book_titles_from_folder(folder_path):
    files = os.listdir(folder_path)
    # 只返回非 .py 文件的完整文件名（包括扩展名）
    book_titles = [file for file in files if os.path.isfile(os.path.join(folder_path, file)) and not file.endswith('.py')]
    return book_titles

# 预定义的英文分类数组
categories = ['Technology', 'Science', 'Fiction', 'History', 'Psychology', 'Economics', 'Science Fiction', 
              'Biography', 'Management', 'Self-help', 'Literature', 'Philosophy', 'Travel', 'Engineering']

# 调用 Google Books API 获取书籍分类信息
def get_category_from_google_books(file_name):
    book_title = os.path.splitext(file_name)[0]  # 去掉扩展名
    url = f"https://www.googleapis.com/books/v1/volumes?q=intitle:{book_title}"

    try:
        response = requests.get(url)
        data = response.json()

        # 检查是否有结果
        if 'items' in data and len(data['items']) > 0:
            # 遍历返回结果
            for item in data['items']:
                if 'categories' in item['volumeInfo']:
                    subjects = item['volumeInfo']['categories']  # 获取分类信息
                    print(f"书名: {book_title}, 返回的分类: {subjects}")
                    
                    # 遍历 subjects，进行分类匹配
                    for subject in subjects:
                        for category in categories:
                            if category.lower() in subject.lower():  # 匹配分类，忽略大小写
                                return category  # 返回匹配到的分类
                    
                    # 如果没有匹配到预定义的分类，返回 subjects 数组的第一个分类
                    return subjects[0]
        return '未分类'  # 如果没有匹配到，则返回未分类
    except Exception as e:
        print(f"调用 Google Books API 失败: {e}")
        return '未分类'

# 将文件移动到相应的文件夹
def move_file_to_category_folder(file_name, source_folder, category_folder):
    if not os.path.exists(category_folder):
        os.makedirs(category_folder)
    source_file = os.path.join(source_folder, file_name)
    target_file = os.path.join(category_folder, file_name)
    shutil.move(source_file, target_file)

# 主函数：调用 Google Books API 进行分类并移动文件
def categorize_and_move_books_google(folder_path):
    book_titles = get_book_titles_from_folder(folder_path)
    
    for title in book_titles:
        category = get_category_from_google_books(title)
        print(f"书名: {title}, 类别: {category}")
        
        # 使用返回的分类或者 '未分类' 进行文件夹的创建和文件移动
        if category != '未分类':  # 如果找到分类，创建相应文件夹并移动
            category_folder = os.path.join(folder_path, category)
            move_file_to_category_folder(title, folder_path, category_folder)
        else:
            print(f"书名: {title} 未能分类，保留在当前文件夹")

# 示例：执行分类并移动文件
folder_path = 'E:/我的小书屋/未分类'  # 书籍所在文件夹
categorize_and_move_books_google(folder_path)

```
